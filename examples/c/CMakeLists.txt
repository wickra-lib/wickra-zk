# Build and run the C and C++ examples and the C ABI golden test against the
# wickra-zk C ABI.
#
#   cargo build --release -p wickra-zk-c
#   cmake -S examples/c -B examples/c/build
#   cmake --build examples/c/build --config Release
#   ctest --test-dir examples/c/build -C Release --output-on-failure
#
# WICKRA_ZK_LIB_DIR defaults to the workspace target/release directory and can
# be overridden for out-of-tree builds. The examples and the test run the
# prover in dev-mode (RISC0_DEV_MODE=1, set by ctest below): a dev-mode receipt
# takes milliseconds, a real one takes minutes, and what is exercised here is
# the envelope, not the proof system.

cmake_minimum_required(VERSION 3.15)
project(wickra_zk_examples C CXX)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(WICKRA_ZK_LIB_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../target/release"
    CACHE PATH "Directory containing the built wickra-zk C ABI library")

set(WICKRA_ZK_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../../bindings/c/include")

find_library(WICKRA_ZK_LIB
    NAMES wickra_zk libwickra_zk
    HINTS "${WICKRA_ZK_LIB_DIR}"
    REQUIRED)

# The golden corpus, globbed at configure time and written into a header, so the
# test covers a case added to the corpus without editing the test. The case ->
# dataset mapping is read from golden/cases.json at run time, the same file
# every other binding's golden test reads. CONFIGURE_DEPENDS makes the build
# system re-glob when the directory changes.
set(GOLDEN_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../golden")
file(GLOB GOLDEN_SPEC_FILES CONFIGURE_DEPENDS "${GOLDEN_DIR}/specs/*.json")
set(GOLDEN_CASE_ENTRIES "")
foreach(spec_file ${GOLDEN_SPEC_FILES})
    get_filename_component(case_name "${spec_file}" NAME_WE)
    string(APPEND GOLDEN_CASE_ENTRIES "    \"${case_name}\",\n")
endforeach()
list(LENGTH GOLDEN_SPEC_FILES GOLDEN_CASE_COUNT)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/golden_cases.h"
"/* Generated by examples/c/CMakeLists.txt -- do not edit. */
#ifndef WICKRA_ZK_GOLDEN_CASES_H
#define WICKRA_ZK_GOLDEN_CASES_H
#include <stddef.h>
#define GOLDEN_DIR \"${GOLDEN_DIR}\"
static const char *GOLDEN_CASES[] = {
${GOLDEN_CASE_ENTRIES}    NULL
};
static const size_t GOLDEN_CASE_COUNT = ${GOLDEN_CASE_COUNT};
#endif
")

add_executable(prove prove.c)
target_include_directories(prove PRIVATE "${WICKRA_ZK_INCLUDE}" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(prove PRIVATE "${WICKRA_ZK_LIB}")

add_executable(provepp prove.cpp)
target_include_directories(provepp PRIVATE "${WICKRA_ZK_INCLUDE}" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(provepp PRIVATE "${WICKRA_ZK_LIB}")

# The test, as opposed to the examples above: `prove` and `provepp` print a
# journal and exit zero. This one asserts golden parity over the whole corpus.
add_executable(golden_test golden_test.c)
target_include_directories(golden_test PRIVATE "${WICKRA_ZK_INCLUDE}" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(golden_test PRIVATE "${WICKRA_ZK_LIB}")

enable_testing()
add_test(NAME prove COMMAND prove)
add_test(NAME provepp COMMAND provepp)
add_test(NAME golden_test COMMAND golden_test)
set_tests_properties(prove provepp golden_test PROPERTIES
    ENVIRONMENT "RISC0_DEV_MODE=1"
    # A missing runtime dependency must fail fast, never block the runner for
    # hours -- and so must a run that somehow reached the real prover.
    TIMEOUT 120)
